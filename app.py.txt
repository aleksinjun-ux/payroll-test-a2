# Инициализация сессии
if "current_index" not in st.session_state:
    st.session_state.current_index = 0
if "answers" not in st.session_state:
    st.session_state.answers = [-1] * len(quiz_data)
if "quiz_finished" not in st.session_state:
    st.session_state.quiz_finished = False

st.title("📊 Тестирование знаний: Бухгалтер по расчету заработной платы (A2/Middle)")

if not st.session_state.quiz_finished:
    idx = st.session_state.current_index
    item = quiz_data[idx]
    
    # Индикатор прогресса
    answered_count = sum(1 for x in st.session_state.answers if x != -1)
    st.progress(answered_count / len(quiz_data))
    st.caption(f"Вопрос {idx + 1} из {len(quiz_data)} | Категория: **{item['cat']}**")
    
    # Текст вопроса и варианты
    st.markdown(f"### {item['q']}")
    
    # Получаем сохраненный ответ
    current_selection = st.session_state.answers[idx]
    
    # Отображение радио-кнопок
    choice = st.radio(
        "Выберите один правильный вариант:", 
        item['opts'], 
        index=current_selection if current_selection != -1 else None,
        key=f"q_radio_{idx}"
    )
    
    # Сохраняем ответ в сессию
    if choice in item['opts']:
        st.session_state.answers[idx] = item['opts'].index(choice)
        
    # Кнопки навигации
    col1, col2, col3 = st.columns([1, 1, 1])
    
    with col1:
        if st.button("⬅️ Назад", disabled=(idx == 0)):
            st.session_state.current_index -= 1
            st.rerun()
            
    with col2:
        if idx < len(quiz_data) - 1:
            if st.button("Дальше ➡️"):
                st.session_state.current_index += 1
                st.rerun()
                
    with col3:
        if idx == len(quiz_data) - 1:
            if st.button("🏁 Завершить тест"):
                if -1 in st.session_state.answers:
                    st.error("Вы ответили не на все вопросы! Вернитесь назад и проверьте пропущенные.")
                else:
                    st.session_state.quiz_finished = True
                    st.rerun()

else:
    # Страница вывода результатов
    st.success("🎉 Тестирование успешно завершено!")
    
    score = 0
    cat_stats = {}
    
    for i, item in enumerate(quiz_data):
        cat = item['cat']
        if cat not in cat_stats:
            cat_stats[cat] = {"correct": 0, "total": 0}
        cat_stats[cat]["total"] += 1
        
        if st.session_state.answers[i] == item['corr']:
            score += 1
            cat_stats[cat]["correct"] += 1
            
    percentage = (score / len(quiz_data)) * 100
    
    st.metric("Общий результат кандидата", f"{score} / {len(quiz_data)} баллов", f"{percentage:.1f}%")
    
    if percentage >= 75:
        st.balloons()
        st.success("**Статус:** Кандидат успешно подтвердил профессиональный уровень (A2 / Middle).")
    else:
        st.warning("**Статус:** Уровень знаний ниже ожидаемого. Требуется обучение или проверка базовых навыков.")
        
    st.subheader("📊 Аналитика по профессиональным блокам")
    for cat, stats in cat_stats.items():
        cat_perc = (stats['correct'] / stats['total']) * 100
        st.write(f"- **{cat}**: {stats['correct']} из {stats['total']} ({cat_perc:.1f}%)")
        
    st.markdown("---")
    st.subheader("🔍 Детальный разбор ошибок (для проверяющего)")
    
    has_errors = False
    for i, item in enumerate(quiz_data):
        user_ans = st.session_state.answers[i]
        if user_ans != item['corr']:
            has_errors = True
            with st.expander(f"❌ Ошибка в вопросе №{i+1} ({item['cat']})"):
                st.write(f"**Вопрос:** {item['q']}")
                st.write(f"**Ответ кандидата:** {item['opts'][user_ans]}")
                st.write(f"**Правильный ответ:** :green[{item['opts'][item['corr']]}]")
                st.info(f"**Нормативное обоснование (РФ):** {item['expl']}")
                
    if not has_errors:
        st.write("Идеально! Кандидат ответил правильно на все 30 вопросов.")
        
    if st.button("🔄 Пройти тест заново"):
        st.session_state.current_index = 0
        st.session_state.answers = [-1] * len(quiz_data)
        st.session_state.quiz_finished = False
        st.rerun()
